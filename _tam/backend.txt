var COT = ['Thời gian đăng ký','Tên','Số điện thoại','Email','Quà','Nguồn','Mã xác nhận','Đã xác nhận','Thời gian xác nhận','Số lần gửi lại'];

/* ============ KHO DỮ LIỆU ============ */
function laySheet() {
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('SHEET_ID');
  if (id) { try { return SpreadsheetApp.openById(id).getSheets()[0]; } catch (e) {} }

  var ss = SpreadsheetApp.create(CAU_HINH.tenSheet);
  var sh = ss.getSheets()[0];
  sh.setName('LEAD');
  sh.getRange(1, 1, 1, COT.length).setValues([COT])
    .setFontWeight('bold').setBackground('#3E5541').setFontColor('#FFFFFF');
  sh.setFrozenRows(1);
  sh.setColumnWidth(1, 150); sh.setColumnWidth(2, 160); sh.setColumnWidth(3, 120);
  sh.setColumnWidth(4, 230); sh.setColumnWidth(5, 130); sh.setColumnWidth(7, 200);
  ss.setSpreadsheetLocale('vi_VN');
  try {
    var it = DriveApp.getFoldersByName(CAU_HINH.thuMuc);
    if (it.hasNext()) DriveApp.getFileById(ss.getId()).moveTo(it.next());
  } catch (e) {}
  props.setProperty('SHEET_ID', ss.getId());
  return sh;
}

function layQua(ma) {
  return QUA[ma] || QUA['QT-001'];
}

/* tìm dòng theo email VÀ mã quà — 1 người có thể xin nhiều quà khác nhau */
function timDong(sh, email, ma) {
  var n = sh.getLastRow();
  if (n < 2) return null;
  var v = sh.getRange(2, 4, n - 1, 2).getValues();   // cột D (Email) + E (Quà)
  for (var i = 0; i < v.length; i++) {
    if (String(v[i][0]).trim().toLowerCase() === email && String(v[i][1]).trim() === ma) return i + 2;
  }
  return null;
}

function timTheoMa(sh, ma) {
  var n = sh.getLastRow();
  if (n < 2) return null;
  var v = sh.getRange(2, 7, n - 1, 1).getValues();
  for (var i = 0; i < v.length; i++) {
    if (String(v[i][0]) === ma) return i + 2;
  }
  return null;
}

function taoMa() {
  return Utilities.getUuid().replace(/-/g, '').substring(0, 20);
}

/* ============ NHẬN ĐĂNG KÝ ============ */
function doPost(e) {
  var kq = { ok: false, thongDiep: '' };
  try {
    var d = JSON.parse(e.postData.contents);
    var ten   = String(d.ten   || '').trim().substring(0, 80);
    var sdt   = String(d.sdt   || '').trim().substring(0, 25);
    var email = String(d.email || '').trim().toLowerCase().substring(0, 120);
    var nguon = String(d.nguon || 'khong-ro').trim().substring(0, 60);
    var maQua = String(d.qua   || 'QT-001').trim().toUpperCase();
    if (!QUA[maQua]) maQua = 'QT-001';

    if (!ten)  { kq.thongDiep = 'Chị cho em xin tên với ạ.'; return traLoi(kq); }
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]{2,}$/.test(email)) {
      kq.thongDiep = 'Email chưa đúng định dạng, chị kiểm tra lại giúp em ạ.'; return traLoi(kq);
    }

    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try {
      var sh = laySheet();
      var dong = timDong(sh, email, maQua);

      if (dong) {
        var daXacNhan = sh.getRange(dong, 8).getValue() === true;
        var maXN = String(sh.getRange(dong, 7).getValue());
        sh.getRange(dong, 10).setValue((Number(sh.getRange(dong, 10).getValue()) || 0) + 1);
        if (daXacNhan) {
          guiMailQua(email, ten, maQua);
          kq.ok = true;
          kq.thongDiep = 'Email này đã xác nhận trước đó rồi ạ — em vừa gửi lại quà vào hộp thư của chị.';
        } else {
          guiMailXacNhan(email, ten, maXN, maQua);
          kq.ok = true;
          kq.thongDiep = 'Em vừa gửi lại thư xác nhận. Chị mở hộp thư và bấm nút xác nhận giúp em nha.';
        }
      } else {
        var maMoi = taoMa();
        sh.appendRow([new Date(), ten, sdt, email, maQua, nguon, maMoi, false, '', 1]);
        guiMailXacNhan(email, ten, maMoi, maQua);
        kq.ok = true;
        kq.thongDiep = 'Xong rồi ạ! Em vừa gửi một email tới hộp thư của chị.';
      }
    } finally {
      lock.releaseLock();
    }
  } catch (err) {
    kq.thongDiep = 'Có trục trặc khi gửi, chị thử lại giúp em sau ít phút ạ.';
  }
  return traLoi(kq);
}

function traLoi(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

/* ============ XÁC NHẬN ============ */
function doGet(e) {
  var maXN = e && e.parameter ? String(e.parameter.t || '') : '';
  if (!maXN) return trangHtml('Liên kết không hợp lệ', 'Chị bấm lại đúng nút trong email giúp em nha.', null);

  var sh = laySheet();
  var dong = timTheoMa(sh, maXN);
  if (!dong) return trangHtml('Không tìm thấy đăng ký', 'Liên kết này không còn đúng. Chị đăng ký lại giúp em ạ.', null);

  var ten   = String(sh.getRange(dong, 2).getValue());
  var email = String(sh.getRange(dong, 4).getValue());
  var maQua = String(sh.getRange(dong, 5).getValue()).trim().toUpperCase();
  if (!QUA[maQua]) maQua = 'QT-001';
  var daXacNhan = sh.getRange(dong, 8).getValue() === true;

  if (!daXacNhan) {
    sh.getRange(dong, 8).setValue(true);
    sh.getRange(dong, 9).setValue(new Date());
    guiMailQua(email, ten, maQua);
  }
  return trangHtml('Xác nhận xong rồi ạ 💛',
    'Em đã gửi ' + layQua(maQua).ten + ' vào hộp thư của chị. Chị mở email là bấm nhận quà được ngay.', maQua);
}

function guiMailQua(email, ten, maQua) {

  var buoc = '';
  for (var i = 0; i < q.buoc.length; i++) {
    buoc += '<p style="margin:0 0 10px;">' + (i + 1) + '. ' + q.buoc[i] + '</p>';
  }
  var html = khung(
    '<p style="margin:0 0 12px;">Chào chị <b>' + ten + '</b>,</p>' +
    '<p style="margin:0 0 12px;">Quà của chị đây ạ — <b>' + q.ten + '</b>.</p>' +
    nut(q.link, 'Nhận quà của chị') +
    '<p style="margin:18px 0 8px;"><b>Chị dùng thế nào:</b></p>' + buoc +
    '<p style="margin:14px 0 0;font-size:13.5px;color:#5A6156;">' + q.ketCau + '</p>');
  MailApp.sendEmail({
    to: email,
    subject: 'Quà của chị đây ạ — ' + q.ten,
    htmlBody: html,
    name: CAU_HINH.nguoiGui
  });
}
