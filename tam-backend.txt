var CAU_HINH = {
  tenSheet : 'LEAD CỔNG QUÀ — Đàm Hoa (xay-kenh)',
  nguoiGui : 'Đàm Hoa',
  linkNhomMacDinh: 'https://zalo.me/g/adohlltts6zacuoyoh0m',
  tenNhom  : 'Cộng Đồng Làm Ảnh/Video AI Bán Hàng – Đàm Hoa',
  BAT_EMAIL: false,      // chị duyệt 2 thư xong mới đổi thành true
  gioNhac  : 24
};

/* 6 cột đầu theo đề bài; các cột sau chỉ THÊM VÀO ĐUÔI */
var COT = ['Ngày','Nguồn','Tên','SĐT/Zalo','Email','Đồng ý',
           'Mã','Thư chúc mừng','Đã bấm vào nhóm','Thư nhắc 24h','Đã vào nhóm (chị tick)','Ngừng nhận (chị tick)','Số lần điền'];
var C = { NGAY:1, NGUON:2, TEN:3, SDT:4, EMAIL:5, DONGY:6, MA:7, THU1:8, BAM:9, THU2:10, VAO:11, NGUNG:12, LAN:13 };

/* ============ KHO DỮ LIỆU ============ */
function laySo() {
  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('SHEET_ID');
  if (id) { try { return SpreadsheetApp.openById(id); } catch (e) {} }
  var ss = SpreadsheetApp.create(CAU_HINH.tenSheet);
  ss.setSpreadsheetLocale('vi_VN');
  var sh = ss.getSheets()[0];
  sh.setName('LEAD');
  sh.getRange(1, 1, 1, COT.length).setValues([COT])
    .setFontWeight('bold').setBackground('#1B2A4A').setFontColor('#FFFFFF');
  sh.setFrozenRows(1);
  [150,90,140,120,220,70,170,150,150,150,130,130,90].forEach(function(w, i){ sh.setColumnWidth(i + 1, w); });
  var cd = ss.insertSheet('CAI DAT');
  cd.getRange('A1:B1').setValues([['Link nhóm Zalo', CAU_HINH.linkNhomMacDinh]]);
  cd.getRange('A2').setValue('Đổi link nhóm: chỉ sửa ô B1. Trang cổng + email tự lấy theo.');
  cd.setColumnWidth(1, 160); cd.setColumnWidth(2, 380);
  props.setProperty('SHEET_ID', ss.getId());
  return ss;
}
function laySheet() { return laySo().getSheetByName('LEAD'); }
function layLinkNhom() {
  try {
    var v = String(laySo().getSheetByName('CAI DAT').getRange('B1').getValue()).trim();
    if (/^https:\/\/zalo\.me\//.test(v)) return v;
  } catch (e) {}
  return CAU_HINH.linkNhomMacDinh;
}

function timTheoCot(sh, cot, giaTri) {
  var n = sh.getLastRow();
  if (n < 2) return null;
  var v = sh.getRange(2, cot, n - 1, 1).getValues();
  for (var i = 0; i < v.length; i++) if (String(v[i][0]).trim().toLowerCase() === giaTri) return i + 2;
  return null;
}
function taoMa() { return Utilities.getUuid().replace(/-/g, '').substring(0, 16); }

/* ============ NHẬN FORM ============ */
function doPost(e) {
  var kq = { ok: false, thongDiep: '' };
  try {
    var d = JSON.parse(e.postData.contents);
    var ten   = String(d.ten   || '').trim().substring(0, 80);
    var sdt   = String(d.sdt   || '').replace(/[^\d]/g, '').substring(0, 15);
    var email = String(d.email || '').trim().toLowerCase().substring(0, 120);
    var nguon = String(d.nguon || 'khong-ro').toLowerCase().replace(/[^a-z0-9-]/g, '').substring(0, 20) || 'khong-ro';
    if (!ten)  { kq.thongDiep = 'Bạn cho Hoa xin tên với nhé.'; return traLoi(kq); }
    if (!/^0\d{9}$/.test(sdt)) { kq.thongDiep = 'Số điện thoại/Zalo chưa đúng, bạn ghi đủ 10 số giúp Hoa nhé.'; return traLoi(kq); }
    if (!/^[^@\s]+@[^@\s]+\.[^@\s]{2,}$/.test(email)) { kq.thongDiep = 'Email chưa đúng, bạn kiểm tra lại giúp Hoa nhé.'; return traLoi(kq); }
    if (d.dongy !== true) { kq.thongDiep = 'Bạn tick ô đồng ý giúp Hoa nhé.'; return traLoi(kq); }

    var lock = LockService.getScriptLock();
    lock.waitLock(20000);
    var ma; kq.daGuiThu = false;
    try {
      var sh = laySheet();
      var dong = timTheoCot(sh, C.EMAIL, email);
      if (dong) {
        /* khách điền lại: không thêm dòng, cập nhật SĐT/tên mới, đếm số lần; KHÔNG gửi lại thư (chống spam) */
        ma = String(sh.getRange(dong, C.MA).getValue());
        sh.getRange(dong, C.TEN).setValue(ten);
        sh.getRange(dong, C.SDT).setValue("'" + sdt);
        sh.getRange(dong, C.LAN).setValue((Number(sh.getRange(dong, C.LAN).getValue()) || 1) + 1);
        kq.daGuiThu = !!sh.getRange(dong, C.THU1).getValue();
      } else {
        ma = taoMa();
        sh.appendRow([new Date(), nguon, ten, "'" + sdt, email, 'Có', ma, '', '', '', false, false, 1]);
        dong = sh.getLastRow();
        sh.getRange(dong, C.VAO, 1, 2).insertCheckboxes();   /* ô tick cho chị, chỉ ở dòng có khách (tick sẵn cả cột sẽ làm appendRow nhảy xuống dòng 1000) */
        if (CAU_HINH.BAT_EMAIL) {
          guiThuChucMung(email, ten, ma);
          sh.getRange(dong, C.THU1).setValue(new Date());
          kq.daGuiThu = true;
        }
      }
    } finally { lock.releaseLock(); }
    kq.ok = true; kq.ma = ma; kq.linkNhom = layLinkNhom();
  } catch (err) {
    kq.thongDiep = 'Có trục trặc khi gửi, bạn thử lại giúp Hoa sau ít phút nhé.';
  }
  return traLoi(kq);
}
function traLoi(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

/* ============ BẤM VÀO NHÓM ============
 * ?bam=<mã>  : trang cổng gọi ngầm khi khách bấm nút (trả về trống)
 * ?vao=<mã>  : nút trong email -> ghi nhận rồi hiện nút vào nhóm (Apps Script không chuyển hướng thẳng được) */
function doGet(e) {
  var p = (e && e.parameter) || {};
  var ma = String(p.bam || p.vao || '').trim().toLowerCase();
  if (ma) {
    try {
      var sh = laySheet(), dong = timTheoCot(sh, C.MA, ma);
      if (dong && !sh.getRange(dong, C.BAM).getValue()) sh.getRange(dong, C.BAM).setValue(new Date());
    } catch (err) {}
  }
  if (p.bam) return ContentService.createTextOutput('');
  var link = layLinkNhom();
  var h = '<!DOCTYPE html><html lang="vi"><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"></head>' +
    '<body style="margin:0;padding:40px 16px;background:#F3ECDF;font-family:Segoe UI,Helvetica Neue,Arial,sans-serif;">' +
    '<div style="max-width:480px;margin:0 auto;background:#FFFDF9;border:1.5px solid #E2D6C0;border-radius:18px;padding:30px 24px;text-align:center;">' +
    '<div style="font-size:44px;">🎁</div>' +
    '<h1 style="font-size:21px;color:#1B2A4A;margin:10px 0;">Quà của bạn đang chờ trong nhóm</h1>' +
    '<p style="font-size:15px;color:#5E6270;line-height:1.6;margin:0;">' + CAU_HINH.tenNhom + '</p>' +
    '<a href="' + link + '" target="_top" style="display:inline-block;margin-top:20px;background:#0068FF;color:#fff;text-decoration:none;' +
    'font-weight:800;font-size:16px;padding:14px 28px;border-radius:12px;">Vào nhóm Zalo nhận quà →</a></div></body></html>';
  return HtmlService.createHtmlOutput(h).setTitle('Vào nhóm nhận quà').addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

/* ============ THƯ NHẮC 24H — chạy theo trigger mỗi giờ ============
 * Gửi khi: đã có thư chúc mừng · qua 24h · chưa bấm vào nhóm · chị chưa tick "Đã vào nhóm" · chưa tick "Ngừng nhận" · chưa nhắc lần nào. */
function guiThuNhac() {
  if (!CAU_HINH.BAT_EMAIL) return;
  var sh = laySheet(), n = sh.getLastRow();
  if (n < 2) return;
  var v = sh.getRange(2, 1, n - 1, COT.length).getValues(), bayGio = new Date();
  for (var i = 0; i < v.length; i++) {
    var r = v[i];
    if (!r[C.THU1 - 1] || r[C.BAM - 1] || r[C.THU2 - 1] || r[C.VAO - 1] === true || r[C.NGUNG - 1] === true) continue;
    if (bayGio - new Date(r[C.THU1 - 1]) < CAU_HINH.gioNhac * 3600 * 1000) continue;
    if (MailApp.getRemainingDailyQuota() < 5) break;
    try {
      guiThuNhac1(String(r[C.EMAIL - 1]), String(r[C.TEN - 1]), String(r[C.MA - 1]));
      sh.getRange(i + 2, C.THU2).setValue(new Date());
    } catch (e) {}
  }
}

/* ============ CHẠY TAY 1 LẦN khi cài ============ */
function khoiTao() {
  var ss = laySo();
  ScriptApp.getProjectTriggers().forEach(function(t){ if (t.getHandlerFunction() === 'guiThuNhac') ScriptApp.deleteTrigger(t); });
  ScriptApp.newTrigger('guiThuNhac').timeBased().everyHours(1).create();
  Logger.log('Sheet: ' + ss.getUrl());
  return ss.getUrl();
}
